{{- /* Resolve author id → display name + profile href. Lab member = content/authors/ with user_groups; the page title (per language) is the single source for display names. lab_author_aliases maps display/alternate names → slug. Returns a dict (name, href) — no Scratch side effects, so the result cannot go stale when the same page is rendered concurrently. */ -}} {{- $raw := printf "%v" .id -}} {{- $id := $raw -}} {{- with hugo.Data.lab_author_aliases -}} {{- with index . $raw -}} {{- $id = . -}} {{- end -}} {{- end -}} {{- $name := $raw -}} {{- $href := "" -}} {{- $slug := $id | urlize -}} {{- with site.GetPage (printf "/authors/%s" $slug) -}} {{- if .File -}} {{- $name = .LinkTitle -}} {{- if isset .Params "user_groups" -}} {{- $href = .RelPermalink -}} {{- end -}} {{- end -}} {{- end -}} {{- return (dict "name" $name "href" $href) -}}
