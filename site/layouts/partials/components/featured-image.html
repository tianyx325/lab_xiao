{{- /* Returns featured image data (src, srcset, width, height) for a page. 16:9 landscape crops — matches the card media slots (poster top band, media-row left zone) and row thumbnails. */ -}} {{- $page := . -}} {{- $result := dict -}} {{- $resource := partial "blox-core/functions/get_featured_image.html" $page -}} {{- $anchor := "Smart" -}} {{- if reflect.IsMap $page.Params.image -}} {{- with $page.Params.image.focal_point -}} {{- $anchor = . -}} {{- end -}} {{- end -}} {{- with $resource -}} {{- $xl := .Fill (printf "960x540 %s webp q74" $anchor) -}} {{- $lg := .Fill (printf "640x360 %s webp q72" $anchor) -}} {{- $md := .Fill (printf "480x270 %s webp q70" $anchor) -}} {{- $sm := .Fill (printf "320x180 %s webp q68" $anchor) -}} {{- $result = dict "src" $md.RelPermalink "srcset" (printf "%s 320w, %s 480w, %s 640w, %s 960w" $sm.RelPermalink $md.RelPermalink $lg.RelPermalink $xl.RelPermalink) "width" $md.Width "height" $md.Height -}} {{- end -}} {{- return $result -}}
