{{/* Build a root-absolute, language-aware menu href. Always forces a leading slash so links never nest under the current page (e.g. /research/ + publications/ → /research/publications/). */}} {{- $u := . | default "/" | string -}} {{- if or (strings.HasPrefix $u "http://") (strings.HasPrefix $u "https://") (strings.HasPrefix $u "//") -}} {{- return $u -}} {{- end -}} {{- if not (strings.HasPrefix $u "/") -}} {{- $u = printf "/%s" $u -}} {{- end -}} {{- return ($u | relLangURL) -}}
