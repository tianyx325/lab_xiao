{{/* Build a root-absolute, language-aware menu href. Always forces a leading slash so links never nest under the current page (e.g. /research/ + publications/ → /research/publications/). */}} {{- $u := . | default "/" | string -}} {{- if or (strings.HasPrefix $u "http://") (strings.HasPrefix $u "https://") (strings.HasPrefix $u "//") -}} {{- return $u -}} {{- end -}} {{- /* Hugo already applies the base path to menu URLs (/lab_xiao/research/), so strip it before calling relLangURL. relLangURL only handles a sub-path baseURL correctly for slash-less input. */ -}} {{- $base := (urls.Parse site.BaseURL).Path -}} {{- if and $base (ne $base "/") -}} {{- $u = strings.TrimPrefix $base $u -}} {{- end -}} {{- $u = strings.TrimPrefix "/" $u -}} {{- return ($u | relLangURL) -}}
